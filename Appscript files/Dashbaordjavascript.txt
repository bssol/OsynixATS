<script>
let dashboardData = null;
let dashboardUser = null;


document.addEventListener(
  'osynixUserReady',
  function(event) {
    dashboardUser =
      event.detail || {};

    dashboardApplyUser_(
      dashboardUser
    );

    dashboardLoad();
  },
  {
    once: true
  }
);


function dashboardSessionToken_() {
  let token = '';

  try {
    token =
      localStorage.getItem(
        'OSYNIX_ATS_SESSION'
      ) || '';
  } catch (error) {
    token = '';
  }

  if (!token) {
    try {
      token =
        new URLSearchParams(
          window.location.search
        ).get(
          'session'
        ) || '';
    } catch (error) {
      token = '';
    }
  }

  if (!token) {
    throw new Error(
      'Your ATS session is missing. Please sign in again.'
    );
  }

  return token;
}


function dashboardApplyUser_(user) {
  const greeting =
    document.getElementById(
      'dashboardGreeting'
    );

  if (greeting) {
    const firstName =
      String(
        user.fullName ||
        user.email ||
        'User'
      )
        .trim()
        .split(/\s+/)[0];

    greeting.textContent =
      dashboardGreetingPrefix_() +
      ', ' +
      firstName;
  }

  const createButton =
    document.getElementById(
      'dashboardCreatePosition'
    );

  if (
    createButton &&
    user.permissions &&
    user.permissions
      .canCreatePositions === false
  ) {
    createButton.style.display =
      'none';
  }
}


function dashboardGreetingPrefix_() {
  const hour =
    new Date().getHours();

  if (hour < 12) {
    return 'Good morning';
  }

  if (hour < 17) {
    return 'Good afternoon';
  }

  return 'Good evening';
}


function dashboardLoad() {
  const loading =
    document.getElementById(
      'dashboardLoading'
    );

  const content =
    document.getElementById(
      'dashboardContent'
    );

  if (loading) {
    loading.textContent =
      'Loading live ATS dashboard...';

    loading.classList.remove(
      'hidden'
    );
  }

  if (content) {
    content.classList.add(
      'hidden'
    );
  }

  google.script.run
    .withSuccessHandler(
      function(result) {
        dashboardData =
          result || {};

        if (
          result &&
          result.user
        ) {
          dashboardUser =
            result.user;

          dashboardApplyUser_(
            dashboardUser
          );
        }

        dashboardRender();

        if (loading) {
          loading.classList.add(
            'hidden'
          );
        }

        if (content) {
          content.classList.remove(
            'hidden'
          );
        }
      }
    )
    .withFailureHandler(
      function(error) {
        if (loading) {
          loading.textContent =
            dashboardErrorMessage_(
              error
            );
        }
      }
    )
    .getRecruiterDashboardData(
      dashboardSessionToken_()
    );
}


function dashboardRefresh() {
  dashboardLoad();
}

}

function dashboardRenderPositionStatus_() {
  const container =
    document.getElementById(
      'dashboardPositionStatus'
    );

  if (!container) {
    return;
  }

  const positions =
    Array.isArray(
      dashboardData.allPositions
    )
      ? dashboardData.allPositions
      : [];

  function countStatus(status) {
    return positions.filter(
      function(item) {
        return (
          dashboardNormalize_(
            item.status
          ) ===
          dashboardNormalize_(
            status
          )
        );
      }
    ).length;
  }

  const items = [
    {
      label: 'Total Positions',
      status: '',
      value: positions.length
    },
    {
      label: 'Active',
      status: 'Active',
      value: countStatus('Active')
    },
    {
      label: 'On Hold',
      status: 'On Hold',
      value: countStatus('On Hold')
    },
    {
      label: 'Filled',
      status: 'Filled',
      value: countStatus('Filled')
    },
    {
      label: 'Closed',
      status: 'Closed',
      value: countStatus('Closed')
    },
    {
      label: 'Cancelled',
      status: 'Cancelled',
      value: countStatus('Cancelled')
    }
  ];

  container.innerHTML =
    items.map(
      function(item) {
        return `
          <button
            class="position-status-card"
            type="button"
            onclick="dashboardGoToManagePositions_('${dashboardEscapeAttr_(

}

function dashboardRenderFunnel_() {
  const container =
    document.getElementById(
      'dashboardFunnel'
    );

  if (!container) {
    return;
  }

  const items =
    Array.isArray(
      dashboardData.funnel
    )
      ? dashboardData.funnel
      : [];

  const max =
    Math.max(
      1,
      ...items.map(
        function(item) {
          return Number(
            item.count || 0
          );
        }
      )
    );

  container.innerHTML =
    items.map(
      function(item) {
        const height =
          Math.max(
            8,
            Math.round(
              Number(
                item.count || 0
              ) /
              max *
              100
            )
          );

        return `
          <div
            class="funnel-item"
            onclick="dashboardOpenFunnel('${dashboardEscapeAttr_(
              item.key
            )}')"
          >
            <span class="funnel-number">
              ${dashboardEscape_(
                item.count || 0
              )}
            </span>

            <div class="funnel-track">
              <div
                class="funnel-bar"
                style="height:${height}%"
              ></div>
            </div>

            <span class="funnel-label">
              ${dashboardEscape_(
                item.label
              )}
            </span>
          </div>
        `;
      }
    )
    .join('');
}


function dashboardRenderDecisionMix_() {
  const container =
    document.getElementById(
      'dashboardDecisionMix'
    );

  if (!container) {
    return;
  }

  const items =
    Array.isArray(
      dashboardData.decisionMix
    )
      ? dashboardData.decisionMix
      : [];

  const total =
    Math.max(
      1,
      items.reduce(
        function(sum, item) {
          return sum +
            Number(
              item.count || 0
            );
        },
        0
      )
    );

  const degrees = [];
  let running = 0;

  items.forEach(
    function(item) {
      running +=
        Number(
          item.count || 0
        ) /
        total *
        360;

      degrees.push(
        Math.round(
          running
        ) +
        'deg'
      );
    }
  );

  while (
    degrees.length < 4
  ) {
    degrees.push(
      '360deg'
    );
  }

  const colors = [
    '#175AFF',
    '#F59E0B',
    '#EF4444',
    '#16A36A'
  ];

  container.innerHTML = `
    <div
      class="donut"
      style="
        --s1:${degrees[0]};
        --s2:${degrees[1]};
        --s3:${degrees[2]};
      "
    >
      <div class="donut-center">
        <strong>
          ${dashboardEscape_(
            total
          )}
        </strong>

        <span>
          decisions
        </span>
      </div>
    </div>

    <div class="legend-list">
      ${items.map(
        function(
          item,
          index
        ) {
          const percent =
            Math.round(
              Number(
                item.count || 0
              ) /
              total *
              100
            );

          return `
            <div
              class="legend-item"
              onclick="dashboardOpenDecision('${dashboardEscapeAttr_(
                item.key
              )}')"
            >
              <span
                class="legend-dot"
                style="background:${colors[index] || '#94A3B8'}"
              ></span>

              <span>
                ${dashboardEscape_(
                  item.label
                )}
              </span>

              <strong>
                ${dashboardEscape_(
                  item.count || 0
                )}
                (${percent}%)
              </strong>
            </div>
          `;
        }
      ).join('')}
    </div>
  `;
}

function dashboardOpenManagePositions() {
  const items =
    Array.isArray(
      dashboardData.allPositions
    )
      ? dashboardData.allPositions
      : [];

  dashboardShowDrawer_(
    'Manage Positions',
    `
      <input
        id="drawerPositionSearch"
        class="drawer-search"
        type="search"

        oninput="dashboardFilterManagePositions()"
      >

  window.dashboardDrawerPositions =

function dashboardFilterManagePositions() {
  const input =
    document.getElementById(
      'drawerPositionSearch'
    );

  const results =
    document.getElementById(
      'drawerPositionResults'
    );

  if (
    !input ||
    !results
  ) {
    return;
  }

  const query =
    dashboardNormalize_(
      input.value
    );

  const items =
    (
      window
        .dashboardDrawerPositions ||
      []
    )
      .filter(
        function(item) {
          const haystack =
            [
              item.refNo,
              item.client,
              item.position,
              item.status,
              item.priority,
              item.location
            ]
              .join(' ')
              .toLowerCase();

          return haystack.includes(
            query
          );
        }
      );

  results.innerHTML =
    dashboardPositionTable_(
      items,
      true
    );
}


function dashboardOpenList(type) {
  if (!dashboardData) {
    return;
  }

  if (type === 'positions') {
    dashboardShowDrawer_(
      'All Active Positions',
      dashboardPositionTable_(
        dashboardData.activePositions ||
        [],
        false
      )
    );

    return;
  }

  let candidates =
    dashboardData.candidates ||
    [];

  let title =
    'Candidate Database — Active Positions';

  if (
    type ===
    'shortlisted'
  ) {
    title =
      'Shortlisted Candidates — Active Positions';

    candidates =
      candidates.filter(
        function(item) {
          return (
            dashboardNormalize_(
              item.stage
            ) === 'shortlisted'
          );
        }
      );
  }

  if (
    type ===
    'interviews'
  ) {
    title =
      'Interview Activity — Active Positions';

    candidates =
      candidates.filter(
        function(item) {
          return !!item.interviewStatus;
        }
      );
  }

  if (
    type ===
    'offers'
  ) {
    title =
      'Offer Activity — Active Positions';

    candidates =
      candidates.filter(
        function(item) {
          const stage =
            dashboardNormalize_(
              item.stage
            );

          return (
            stage ===
              'offer sent' ||
            stage ===
              'offered' ||
            stage ===
              'offer'
          );
        }
      );
  }

  if (
    type ===
    'hired'
  ) {
    title =
      'Hired Candidates — Active Positions';

    candidates =
      candidates.filter(
        function(item) {
          return (
            dashboardNormalize_(
              item.stage
            ) === 'hired'
          );
        }
      );
  }

  dashboardShowCandidateDatabase_(
    title,
    candidates
  );
}

function dashboardFilterDrawerCandidates() {
  const input =
    document.getElementById(
      'drawerCandidateSearch'
    );

  const results =
    document.getElementById(
      'drawerCandidateResults'
    );

  if (
    !input ||
    !results
  ) {
    return;
  }

  const query =
    dashboardNormalize_(
      input.value
    );

  const items =
    (
      window
        .dashboardDrawerCandidates ||
      []
    )
      .filter(
        function(item) {
          const haystack =
            [
              item.name,
              item.email,
              item.phone,
              item.client,
              item.position,
              item.refNo,
              item.company,
              item.designation
            ]
              .join(' ')
              .toLowerCase();

          return haystack.includes(
            query
          );
        }
      );

  results.innerHTML =
    dashboardCandidateTable_(
      items
    );
}

function dashboardSavePosition(
  refNo
) {
  const statusEl =
    document.getElementById(
      'managePositionStatus'
    );

  const priorityEl =
    document.getElementById(
      'managePositionPriority'
    );

  const targetEl =
    document.getElementById(
      'managePositionTarget'
    );

  const button =
    document.getElementById(
      'managePositionSaveButton'
    );

  const statusMessage =
    document.getElementById(
      'managePositionSaveStatus'
    );

  if (
    !statusEl ||
    !priorityEl ||
    !targetEl
  ) {
    return;
  }

  const hiringTarget =
    Number(
      targetEl.value
    );

  if (
    !Number.isInteger(
      hiringTarget
    ) ||
    hiringTarget < 1
  ) {
    if (statusMessage) {
      statusMessage.textContent =
        'Hiring Target must be a whole number of 1 or more.';

      statusMessage.className =
        'position-save-status error';
    }

    return;
  }

  if (button) {
    button.disabled =
      true;

    button.textContent =
      'Saving...';
  }

  if (statusMessage) {
    statusMessage.textContent =
      'Updating live ATS...';

    statusMessage.className =
      'position-save-status';
  }

  const payload = {
    refNo:
      refNo,

    status:
      statusEl.value,

    priority:
      priorityEl.value,

    hiringTarget:
      hiringTarget
  };

  google.script.run
    .withSuccessHandler(
      function(result) {
        if (statusMessage) {
          statusMessage.textContent =
            result &&
            result.message
              ? result.message
              : 'Position updated successfully.';

          statusMessage.className =
            'position-save-status success';
        }

        // Reload all live dashboard data so the
        // active-position view responds immediately.
        dashboardCloseDrawer();
        dashboardLoad();
      }
    )
    .withFailureHandler(
      function(error) {
        if (button) {
          button.disabled =
            false;

          button.textContent =
            'Save Changes';
        }

        if (statusMessage) {
          statusMessage.textContent =
            dashboardErrorMessage_(
              error
            );

          statusMessage.className =
            'position-save-status error';
        }
      }
    )
    .updatePositionManagement(
      dashboardSessionToken_(),
      payload
    );
}

function dashboardOpenPending(key) {
  if (
    key ===
      'zero-screened' ||
    key ===
      'high-priority'
  ) {
    let positions =
      dashboardData.activePositions ||
      [];

    if (
      key ===
      'zero-screened'
    ) {
      positions =
        positions.filter(
          function(item) {
            return (
              item.screened === 0
            );
          }
        );
    }

    if (
      key ===
      'high-priority'
    ) {
      positions =
        positions.filter(
          function(item) {
            return (
              dashboardNormalize_(
                item.priority
              ) === 'high'
            );
          }
        );
    }

    dashboardShowDrawer_(
      key ===
        'zero-screened'
        ? 'Positions with Zero Screened'
        : 'High-Priority Positions',
      dashboardPositionTable_(
        positions,
        false
      )
    );

    return;
  }

  let candidates =
    dashboardData.candidates ||
    [];

  if (key === 'hold') {
    candidates =
      candidates.filter(
        function(item) {
          return (
            dashboardNormalize_(
              item.stage
            ) === 'hold'
          );
        }
      );
  }

  if (
    key ===
    'shortlisted-no-interview'
  ) {
    candidates =
      candidates.filter(
        function(item) {
          return (
            dashboardNormalize_(
              item.stage
            ) ===
              'shortlisted' &&
            !item.interviewStatus
          );
        }
      );
  }

  if (
    key ===
    'client-status'
  ) {
    candidates =
      candidates.filter(
        function(item) {
          return (
            dashboardNormalize_(
              item.clientStatus
            ) === 'pending'
          );
        }
      );
  }

  if (key === 'offers') {
    candidates =
      candidates.filter(
        function(item) {
          const stage =
            dashboardNormalize_(
              item.stage
            );

          return (
            stage ===
              'offer sent' ||
            stage ===
              'offered' ||
            stage ===
              'offer'
          );
        }
      );
  }

  dashboardShowCandidateDatabase_(
    'Pending Action — Active Positions',
    candidates
  );
}

function dashboardPriorityClass_(
  priority
) {
  const clean =
    dashboardNormalize_(
      priority
    );

  if (
    clean === 'high'
  ) {
    return 'priority-high';
  }

  if (
    clean === 'low'
  ) {
    return 'priority-low';
  }

  return 'priority-medium';
}


function dashboardStatusClass_(
  status
) {
  const clean =
    dashboardNormalize_(
      status
    );

  if (
    clean === 'active'
  ) {
    return 'status-active';
  }

  if (
    clean === 'on hold'
  ) {
    return 'status-hold';
  }

  if (
    clean === 'filled'
  ) {
    return 'status-filled';
  }

  if (
    clean === 'cancelled'
  ) {
    return 'status-cancelled';
  }

  return 'status-closed';
}


function dashboardNormalize_(value) {
  return String(
    value || ''
  )
    .trim()
    .toLowerCase();
}


function dashboardInitials_(name) {
  const clean =
    String(
      name || ''
    ).trim();

  if (!clean) {
    return '—';
  }

  return clean
    .split(/\s+/)
    .slice(
      0,
      2
    )
    .map(
      function(part) {
        return part
          .charAt(0)
          .toUpperCase();
      }
    )
    .join('');
}


function dashboardErrorMessage_(error) {
  if (!error) {
    return (
      'Dashboard could not be loaded.'
    );
  }

  return (
    error.message ||
    String(
      error
    )
  );
}


function dashboardEscape_(value) {
  return String(
    value ?? ''
  )
    .replace(
      /&/g,
      '&amp;'
    )
    .replace(
      /</g,
      '&lt;'
    )
    .replace(
      />/g,
      '&gt;'
    )
    .replace(
      /"/g,
      '&quot;'
    )
    .replace(
      /'/g,
      '&#039;'
    );
}


function dashboardEscapeAttr_(value) {
  return dashboardEscape_(
    value
  )
    .replace(
      /\r/g,
      ' '
    )
    .replace(
      /\n/g,
      ' '
    );
}
