function restoreBatchAssessmentState() {

      results.length === 0

      return;

      results;

        results

 

      );

      'Could not restore batch assessment state.',

 

      error

    sessionStorage.removeItem(

 

      'osynixBatchAssessmentResults'

 

    );

function saveBatchToATS() {

 

 

 

  const results =

 

    window.currentBatchResults || [];

 

 

 

  const validResults =

    results.filter(

      result =>

        result &&

        !result.error &&

        !result._atsSaved

    );

 

 

 

  if (!validResults.length) {

 

    return showError(

 

      'No unsaved batch assessments are available to save.'

 

    );

 

  }

 

 

 

 

 

  const missingBatchCv =

 

    validResults.some(

 

      assessment =>

 

        !Number.isInteger(

 

          assessment.batchOriginalIndex

 

        ) ||

 

        !currentBatchCvUploads[

 

          assessment.batchOriginalIndex

 

        ] ||

 

        !currentBatchCvUploads[

 

          assessment.batchOriginalIndex

 

        ].fileBase64

 

    );

 

 

 

  if (missingBatchCv) {

 

    return showError(

 

      'The original batch CV files are no longer available in this browser session. Please select and assess the CVs again before saving the batch to ATS.'

 

    );

 

  }

 

 

 

 

 

  const button =

 

    document.getElementById(

 

      'saveBatchBtn'

 

    );

 

 

 

  if (button) {

 

    button.disabled = true;

 

    button.textContent =

 

      'Saving Batch...';

 

  }

 

 

 

 

 

  let currentIndex = 0;

 

  let savedCount = 0;

 

  let failedCount = 0;

 

 

 

 

 

  function saveNextCandidate() {

 

 

 

    /*

 

     * All candidates processed.

 

     */

 

    if (

 

      currentIndex >=

 

      validResults.length

 

    ) {

 

 

 

      if (button) {

 

        button.textContent =

 

          'Batch Saved';

 

 

 

        /*

 

         * Keep disabled after completion

 

         * to prevent accidental duplicate saves.

 

         */

 

        button.disabled = true;

 

      }

 

 

 

 

 

      if (failedCount === 0) {

 

        const allBatchResults =

          window.currentBatchResults || [];

 

        const allValidSaved =

          allBatchResults

            .filter(

              result =>

                result &&

                !result.error

            )

            .every(

              result =>

                result._atsSaved

            );

 

        if (allValidSaved) {

          clearBatchAssessmentState_();

        } else {

          renderBatchResults(

            allBatchResults

          );

        }

        showSuccess(

 

          savedCount +

 

          ' of ' +

 

          validResults.length +

 

          ' candidates saved successfully.'

 

        );

 

 

 

      } else {

 

 

 

        showError(

 

          savedCount +

 

          ' of ' +

 

          validResults.length +

 

          ' candidates saved successfully. ' +

 

          failedCount +

 

          ' failed to save.'

 

        );

 

      }

 

 

 

      return;

 

    }

 

 

 

 

 

    const assessment =

 

      validResults[currentIndex];

 

 

 

 

 

    if (button) {

 

      button.textContent =

 

        'Saving ' +

 

        (currentIndex + 1) +

 

        ' of ' +

 

        validResults.length +

 

        '...';

 

    }

 

 

 

 

 

    google.script.run

 

 

 

      .withSuccessHandler(

 

        function() {

 

          assessment._atsSaved =

            true;

 

          try {

            sessionStorage.setItem(

              'osynixBatchAssessmentResults',

              JSON.stringify(

                window.currentBatchResults || []

              )

            );

          } catch (error) {

            console.warn(

              'Could not update batch save state.',

              error

            );

          }

 

          savedCount++;

          currentIndex++;

 

          saveNextCandidate();

 

        }

 

      )

 

 

 

      .withFailureHandler(

 

        function(error) {

 

 

 

          console.error(

 

            'Batch save failed:',

 

            error

 

          );

 

 

 

          failedCount++;

 

          currentIndex++;

 

 

 

          /*

 

           * Continue with the remaining

 

           * candidates even if one fails.

 

           */

 

          saveNextCandidate();

 

        }

 

      )

 

 

 

      .saveAssessment(

 

  getATSSessionToken_(),

 

  assessment,

 

  currentBatchCvUploads[

 

    assessment.batchOriginalIndex

 

  ]

 

);

 

  }

 

 

 

 

 

  saveNextCandidate();

 

}

function downloadBatchPdf() {

 

 

 

  const results =

 

    window.currentBatchResults || [];

 

 

 

 

 

  const validResults =

 

    results.filter(

 

      result =>

 

        result &&

 

        !result.error

 

    );

 

 

 

 

 

  if (!validResults.length) {

 

    return showError(

 

      'No completed batch assessments are available.'

 

    );

 

  }

 

 

 

 

 

  const button =

 

    document.getElementById(

 

      'downloadBatchPdfBtn'

 

    );

 

 

 

 

 

  if (button) {

 

    button.disabled = true;

 

    button.textContent =

 

      'Preparing Batch PDF...';

 

  }

 

 

 

 

 

  google.script.run

 

 

 

    .withSuccessHandler(

 

      result => {

 

 

 

        const binary =

 

          atob(

 

            result.base64

 

          );

 

 

 

 

 

        const bytes =

 

          new Uint8Array(

 

            binary.length

 

          );

 

 

 

 

 

        for (

 

          let i = 0;

 

          i < binary.length;

 

          i++

 

        ) {

 

          bytes[i] =

 

            binary.charCodeAt(i);

 

        }

 

 

 

 

 

        const blob =

 

          new Blob(

 

            [bytes],

 

            {

 

              type:

 

                result.mimeType ||

 

                'application/pdf'

 

            }

 

          );

 

 

 

 

 

        const url =

 

          URL.createObjectURL(

 

            blob

 

          );

 

 

 

 

 

        const link =

 

          document.createElement(

 

            'a'

 

          );

 

 

 

 

 

        link.href =

 

          url;

 

 

 

 

 

        link.download =

 

          result.fileName ||

 

          'Osynix_Batch_Candidate_Assessment.pdf';

 

 

 

 

 

        document.body.appendChild(

 

          link

 

        );

 

 

 

 

 

        link.click();

 

 

 

 

 

        document.body.removeChild(

 

          link

 

        );

 

 

 

 

 

        URL.revokeObjectURL(

 

          url

 

        );

 

 

 

 

 

        if (button) {

 

          button.disabled = false;

 

          button.textContent =

 

            'Download Batch PDF';

 

        }

 

      }

 

    )

 

 

 

    .withFailureHandler(

 

      error => {

 

 

 

        if (button) {

 

          button.disabled = false;

 

          button.textContent =

 

            'Download Batch PDF';

 

        }

 

 

 

 

 

        showError(

 

          error

 

        );

 

      }

 

    )

 

 

 

    .generateBatchCandidatePdf(

 

      getATSSessionToken_(),

 

      validResults

 

    );

 

}

function renderClientDropdown() {

 

  const select =

 

    document.getElementById(

 

      'clientSelect'

 

    );

 

 

 

  const activePositions =

 

    (bootstrap.positions || []).filter(

 

      p =>

 

        String(p.status || 'Active')

 

          .trim()

 

          .toLowerCase() === 'active'

 

    );

 

 

 

  const clientMap =

 

    new Map();

 

 

 

  activePositions.forEach(position => {

 

    const client =

 

      String(position.client || '')

 

        .trim();

 

 

 

    if (!client) return;

 

 

 

    const key =

 

      normalizeClientName(client);

 

 

 

    if (!clientMap.has(key)) {

 

      clientMap.set(

 

        key,

 

        client

 

      );

 

    }

 

  });

 

 

 

  const clients =

 

    Array.from(

 

      clientMap.values()

 

    ).sort(

 

      (a, b) =>

 

        a.localeCompare(b)

 

    );

 

 

 

  select.innerHTML =

 

    '<option value="">Select a client</option>' +

 

    clients.map(

 

      client => `

 

        <option value="${escapeHtml(client)}">

 

          ${escapeHtml(client)}

 

        </option>

 

      `

 

    ).join('');

 

}

function downloadSinglePdf() {

 

  if (!currentAssessment) {

 

    return showError(

 

      'No candidate assessment is available.'

 

    );

 

  }

 

 

 

  const button =

 

    document.getElementById(

 

      'downloadSinglePdfBtn'

 

    );

 

 

 

  if (button) {

 

    button.disabled = true;

 

    button.textContent =

 

      'Preparing PDF...';

 

  }

 

 

 

  google.script.run

 

 

 

    .withSuccessHandler(

 

      result => {

 

 

 

        const binary =

 

          atob(

 

            result.base64

 

          );

 

 

 

        const bytes =

 

          new Uint8Array(

 

            binary.length

 

          );

 

 

 

        for (

 

          let i = 0;

 

          i < binary.length;

 

          i++

 

        ) {

 

          bytes[i] =

 

            binary.charCodeAt(i);

 

        }

 

 

 

        const blob =

 

          new Blob(

 

            [bytes],

 

            {

 

              type:

 

                result.mimeType ||

 

                'application/pdf'

 

            }

 

          );

 

 

 

        const url =

 

          URL.createObjectURL(

 

            blob

 

          );

 

 

 

        const link =

 

          document.createElement(

 

            'a'

 

          );

 

 

 

        link.href = url;

 

 

 

        link.download =

 

          result.fileName ||

 

          'Osynix_Candidate_Assessment.pdf';

 

 

 

        document.body.appendChild(

 

          link

 

        );

 

 

 

        link.click();

 

 

 

        document.body.removeChild(

 

          link

 

        );

 

 

 

        URL.revokeObjectURL(

 

          url

 

        );

 

 

 

        if (button) {

 

          button.disabled = false;

 

          button.textContent =

 

            'Download PDF Report';

 

        }

 

      }

 

    )

 

 

 

    .withFailureHandler(

 

      error => {

 

 

 

        if (button) {

 

          button.disabled = false;

 

          button.textContent =

 

            'Download PDF Report';

 

        }

 

 

 

        showError(error);

 

      }

 

    )

 

 

 

    .generateSingleCandidatePdf(

 

      getATSSessionToken_(),

 

      currentAssessment

 

    );

 

}

 

 

 

function save() {

 

  if (!currentAssessment) {

    return;

  }

 

  const button =

    document.getElementById(

      'saveBtn'

    );

 

  const isBatchCandidate =

    currentAssessmentSource === 'batch' &&

    Number.isInteger(

      currentViewedBatchIndex

    );

 

  let cvUpload =

    currentCvUpload;

 

  if (isBatchCandidate) {

 

    const batchAssessment =

      (

        window.currentBatchResults ||

        []

      )[currentViewedBatchIndex];

 

    if (

      !batchAssessment ||

      batchAssessment._atsSaved

    ) {

      return showError(

        'This Batch candidate has already been saved to ATS.'

      );

    }

 

    cvUpload =

      currentBatchCvUploads[

        batchAssessment.batchOriginalIndex

      ];

 

    if (

      !cvUpload ||

      !cvUpload.fileBase64

    ) {

      return showError(

        'The original CV for this Batch candidate is no longer available in this browser session. Please reassess the Batch before saving.'

      );

    }

  }

 

  button.disabled = true;

 

  google.script.run

 

    .withSuccessHandler(

      result => {

 

        alert(

          'Saved successfully. ' +

          'Candidate ID: ' +

          result.candidateId

        );

 

        if (isBatchCandidate) {

 

          const batchResults =

            window.currentBatchResults ||

            [];

 

          const savedAssessment =

            batchResults[

              currentViewedBatchIndex

            ];

 

          if (savedAssessment) {

            savedAssessment._atsSaved =

              true;

          }

 

          const allValidSaved =

            batchResults

              .filter(

                item =>

                  item &&

                  !item.error

              )

              .every(

                item =>

                  item._atsSaved

              );

 

          currentAssessment = null;

          currentCvUpload = null;

          currentAssessmentSource = 'single';

          currentViewedBatchIndex = null;

 

          document

            .getElementById(

              'resultCard'

            )

            .classList

            .add('hidden');

 

          document

            .getElementById(

              'result'

            )

            .innerHTML = '';

 

          if (allValidSaved) {

            clearBatchAssessmentState_();

          } else {

            renderBatchResults(

              batchResults

            );

          }

 

        } else {

 

          resetAssessment();

        }

 

        button.disabled = false;

      }

    )

 

    .withFailureHandler(

      error => {

 

        button.disabled = false;

 

        showError(error);

      }

    )

 

    .saveAssessment(

      getATSSessionToken_(),

      currentAssessment,

      cvUpload

    );

}

    jdUnavailable.classList.add(

 

      'hidden'

 

    );

 

  } else {

 

    jdButton.href =

 

      '#';

 

 

 

    jdButton.classList.add(

 

      'hidden'

 

    );

 

 

 

    jdUnavailable.classList.remove(

 

      'hidden'

 

    );

 

  }

 

 

 

 

 

  const criteriaContainer =

 

    document.getElementById(

 

      'sideCriteria'

 

    );

 

 

 

  const list =

 

    Array.isArray(criteria)

 

      ? criteria

 

      : [];

 

 

 

 

 

  if (!list.length) {

 

    criteriaContainer.innerHTML = `

 

      <div class="jd-unavailable">

 

        No approved criteria available.

 

      </div>

 

    `;

 

 

 

    return;

 

  }

 

 

 

 

 

  criteriaContainer.innerHTML =

 

    list.map(

 

      function(item) {

 

 

 

        const criterion =

 

          item['Criterion'] ||

 

          item.criterion ||

 

          'Criterion';

 

 

 

        const weight =

 

          item['Weight %'] ??

 

          item.weight ??

 

          '';

 

 

 

        const knockoutRaw =

 

          item['Knockout?'] ??

 

          item.knockout ??

 

          '';

 

 

 

        const knockout =

 

          String(

 

            knockoutRaw

 

          )

 

            .trim()

 

            .toLowerCase();

 

 

 

        const isKnockout =

 

          knockout === 'yes' ||

 

          knockout === 'true' ||

 

          knockout === '1';

 

 

 

 

 

        return `

 

          <div class="side-criterion-row">

 

 

 

            <div class="side-criterion-name">

 

              ${escapeHtml(criterion)}

 

            </div>

 

 

 

            <div class="side-criterion-weight">

 

              ${escapeHtml(String(weight))}%

 

            </div>

 

 

 

            <div

 

              class="side-knockout ${

 

                isKnockout

 

                  ? 'yes'

 

                  : 'no'

 

              }"

 

            >

 

              ${

 

                isKnockout

 

                  ? 'Yes'

 

                  : 'No'

 

              }

 

            </div>

 

 

 

          </div>

 

        `;

 

      }

 

    )

 

    .join('');

 

}
