function getATSSessionToken_() {
  if (atsSessionTokenCache) {
    return atsSessionTokenCache;
  }

  let token = '';

  try {
    token =
      localStorage.getItem(
        'OSYNIX_ATS_SESSION'
      ) || '';
  } catch (error) {
    token = '';
  }

  if (!token) {
    try {
      const params =
        new URLSearchParams(
          window.location.search
        );

      token =
        params.get(
          'session'
        ) || '';
    } catch (error) {
      token = '';
    }
  }

  if (token) {
    atsSessionTokenCache =
      token;

    try {
      localStorage.setItem(
        'OSYNIX_ATS_SESSION',
        token
      );
    } catch (error) {
      // Keep the in-memory token even if browser storage is unavailable.
    }

    return token;
  }

  throw new Error(
    'Your ATS session is missing. Please sign in again.'
  );
}

function checkNewClientSimilarity() {
  const input =
    document.getElementById(
      'newClientName'
    ).value.trim();

  const warning =
    document.getElementById(
      'clientDuplicateWarning'
    );

  if (!input) {
    clearClientDuplicateWarning();
    return;
  }

  const exact =
    existingClients.find(
      function(client) {
        return normalizeClientName(
          client
        ) ===
        normalizeClientName(
          input
        );
      }
    );

  if (exact) {
    warning.innerHTML = `
      <strong>This client already exists:</strong>
      ${escapeHtml(exact)}
      <br>
      Please select the existing client instead.
    `;

    warning.classList.remove(
      'hidden'
    );

    return;
  }

  const similar =
    existingClients
      .map(
        function(client) {
          return {
            client: client,
            score:
              clientSimilarity(
                input,
                client
              )
          };
        }
      )
      .filter(
        function(item) {
          return item.score >= 0.72;
        }
      )
      .sort(
        function(a, b) {
          return b.score -
            a.score;
        }
      )
      .slice(0, 3);

  if (similar.length) {
    warning.innerHTML = `
      <strong>Possible existing client:</strong>
      ${similar
        .map(
          function(item) {
            return escapeHtml(
              item.client
            );
          }
        )
        .join(', ')}
      <br>
      Please check before creating a new client.
    `;

    warning.classList.remove(
      'hidden'
    );

    return;
  }

  clearClientDuplicateWarning();
}

function normalizeClientName(value) {
  return String(value || '')
    .toLowerCase()
    .replace(
      /[^a-z0-9]+/g,
      ' '
    )
    .replace(
      /\s+/g,
      ' '
    )
    .trim();
}


function clientSimilarity(a, b) {
  const left =
    normalizeClientName(a);

  const right =
    normalizeClientName(b);

  if (!left || !right) {
    return 0;
  }

  if (left === right) {
    return 1;
  }

  if (
    left.includes(right) ||
    right.includes(left)
  ) {
    return 0.9;
  }

  const distance =
    levenshteinDistance(
      left,
      right
    );

  return (
    1 -
    distance /
      Math.max(
        left.length,
        right.length
      )
  );
}


function levenshteinDistance(a, b) {
  const matrix = [];

  for (
    let i = 0;
    i <= b.length;
    i++
  ) {
    matrix[i] = [i];
  }

  for (
    let j = 0;
    j <= a.length;
    j++
  ) {
    matrix[0][j] = j;
  }

  for (
    let i = 1;
    i <= b.length;
    i++
  ) {
    for (
      let j = 1;
      j <= a.length;
      j++
    ) {
      if (
        b.charAt(i - 1) ===
        a.charAt(j - 1)
      ) {
        matrix[i][j] =
          matrix[i - 1][j - 1];
      }
      else {
        matrix[i][j] =
          Math.min(
            matrix[i - 1][j - 1] + 1,
            matrix[i][j - 1] + 1,
            matrix[i - 1][j] + 1
          );
      }
    }
  }

  return matrix[b.length][a.length];
}

  function addCriterion() {
    const existingWeights = getWeightInputs();

    if (existingWeights.length >= 10) {
      alert('A position can contain a maximum of 10 criteria.');
      return;
    }

    const newWeight = existingWeights.length ? 10 : 100;

    if (existingWeights.length) {
      scaleWeightInputsToTotal(existingWeights, 100 - newWeight);
    }

    appendCriterionCard({
      criterion: '',
      weight_percent: newWeight,
      knockout: 'No',
      evidence_expected: '',
      scoring_guidance: '',
      notes: '',
      rationale: 'Recruiter-added criterion. No AI rationale was generated for this item.'
    });

    fixWeightRounding();
    updateWeightTotal();
  }

  function removeCriterion(button) {
    const cards = document.querySelectorAll('.criterion-card');

    if (cards.length <= 4) {
      alert('A position must have at least 4 criteria.');
      return;
    }

    button.closest('.criterion-card').remove();

    const remaining = getWeightInputs();
    scaleWeightInputsToTotal(remaining, 100);
    fixWeightRounding();
    updateWeightTotal();
  }

  function scaleWeightInputsToTotal(inputs, targetTotal) {
    if (!inputs.length) return;

    let currentTotal = inputs.reduce(function(sum, input) {
      return sum + Math.max(0, Number(input.value || 0));
    }, 0);

    if (currentTotal <= 0) {
      const equal = targetTotal / inputs.length;

      inputs.forEach(function(input) {
        input.value = Math.max(1, Math.round(equal));
      });

      return;
    }

    inputs.forEach(function(input) {
      const current = Math.max(0, Number(input.value || 0));
      const scaled = (current / currentTotal) * targetTotal;
      input.value = Math.max(1, Math.round(scaled));
    });
  }

  function fixWeightRounding() {
    const inputs = getWeightInputs();

    if (!inputs.length) return;

    const total = inputs.reduce(function(sum, input) {
      return sum + Number(input.value || 0);
    }, 0);

    const difference = 100 - total;

    if (difference === 0) return;

    const sorted = [...inputs].sort(function(a, b) {
      return Number(b.value || 0) - Number(a.value || 0);
    });

    const target = sorted[0];

    target.value = Math.max(
      1,
      Number(target.value || 0) + difference
    );
  }

  function getWeightInputs() {
    return Array.from(
      document.querySelectorAll('.criterion-weight')
    );
  }

  function collectCriteria() {
    const cards = document.querySelectorAll('.criterion-card');

    return Array.from(cards).map(function(card) {
      return {
        criterion:
          card.querySelector('.criterion-name').value.trim(),

        weight_percent:
          Number(card.querySelector('.criterion-weight').value || 0),

        knockout:
          card.querySelector('.criterion-knockout').value,

        evidence_expected:
          card.querySelector('.criterion-evidence').value.trim(),

        scoring_guidance:
          card.querySelector('.criterion-guidance').value.trim(),

        notes:
          card.querySelector('.criterion-notes').value.trim(),

        rationale:
          card.querySelector('.criterion-rationale').value.trim()
      };
    });
  }

  function updateWeightTotal() {
    const weights = getWeightInputs();

    let total = 0;

    weights.forEach(function(input) {
      total += Number(input.value || 0);
    });

    const label = document.getElementById('weightTotal');
    label.textContent = total + '%';

    const warning = document.getElementById('weightWarning');

    warning.classList.remove('hidden', 'ok', 'bad');

    if (total === 100) {
      label.style.color = '#15803D';
      warning.classList.add('ok');
      warning.textContent = 'Weight allocation is valid. Total = 100%.';
    }
    else {
      label.style.color = '#B42318';
      warning.classList.add('bad');

      if (total > 100) {
        warning.textContent =
          'Total weight is ' +
          total +
          '%. This exceeds 100%. Please reduce one or more criterion weights before saving.';
      }
      else {
        warning.textContent =
          'Total weight is ' +
          total +
          '%. Please allocate the remaining ' +
          (100 - total) +
          '% before saving.';
      }
    }
  }

  function createApprovedPosition() {
    clearStatus('createStatus');

    const confirmed =
      document.getElementById('confirmReview').checked;

    if (!confirmed) {
      showStatus(
        'createStatus',
        'Please confirm that you reviewed the position and JD criteria.',
        'error'
      );
      return;
    }

    const client =
      document.getElementById('reviewClient').value.trim();

    const position =
      document.getElementById('reviewPosition').value.trim();

    const location =
      document.getElementById('reviewLocation').value.trim();

    const criteria = collectCriteria();

    if (!client || !position) {
      showStatus(
        'createStatus',
        'Client and Position Title are required.',
        'error'
      );
      return;
    }

    if (criteria.length < 4 || criteria.length > 10) {
      showStatus(
        'createStatus',
        'Please keep the position between 4 and 10 criteria.',
        'error'
      );
      return;
    }

    const totalWeight = criteria.reduce(function(sum, item) {
      return sum + Number(item.weight_percent || 0);
    }, 0);

    if (totalWeight !== 100) {
      showStatus(
        'createStatus',
        'Criterion weights must total exactly 100%. Current total: ' +
          totalWeight +
          '%.',
        'error'
      );
      return;
    }

    const invalidCriterion = criteria.find(function(item) {
      return (
        !item.criterion ||
        !item.evidence_expected ||
        !item.scoring_guidance
      );
    });

    if (invalidCriterion) {
      showStatus(
        'createStatus',
        'Every criterion needs a name, Evidence Expected and Scoring Guidance. Open View logic for any incomplete criterion.',
        'error'
      );
      return;
    }

    setCreateBusy(true);

    const payload = {
      client: client,
      position: position,
      location: location,
      priority: document.getElementById('priority').value,
      hiringTarget:
        Number(document.getElementById('hiringTarget').value || 1),
      notes:
        document.getElementById('notes').value.trim(),
      criteria: criteria,
      sessionToken:
        getATSSessionToken_()
    };

    if (!uploadedJDFile) {
  showStatus(
    'createStatus',
    'The original JD file is no longer available. Please upload and analyze the JD again.',
    'error'
  );

  setCreateBusy(false);
  return;
}

  function setAnalyzeBusy(isBusy) {
    const button =
      document.getElementById('analyzeBtn');

    button.disabled = isBusy;

    button.textContent =
      isBusy
        ? 'Analyzing JD...'
        : 'Analyze JD';

    if (isBusy) {
      showStatus(
        'analysisStatus',
        'Reading the JD and preparing draft screening criteria...',
        'info'
      );
    }
  }

  function setCreateBusy(isBusy) {
    const button =
      document.getElementById('createBtn');

    button.disabled = isBusy;

    button.textContent =
      isBusy
        ? 'Creating Position...'
        : 'Create Position in Live ATS';
  }

  function showStatus(elementId, message, type) {
    const element =
      document.getElementById(elementId);

    element.textContent = message;

    element.className =
      'status ' +
      (
        type === 'error'
          ? 'error'
          : 'info'
      );
  }

  function clearStatus(elementId) {
    const element =
      document.getElementById(elementId);

    element.textContent = '';
    element.className = 'status hidden';
  }

  function getErrorMessage(error) {
    if (!error) {
      return 'An unknown error occurred.';
    }

    return (
      error.message ||
      String(error)
    );
  }

  function escapeHtml(value) {
    return String(value ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#039;');
  }
