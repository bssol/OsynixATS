  <script>

    function returnToCandidateDatabase() {

      const sessionToken =
        localStorage.getItem(
          'OSYNIX_ATS_SESSION'
        );


      const baseUrl =
        '<?= getWebAppUrl(); ?>';


      window.top.location.href =
        baseUrl +
        '?page=candidatedatabase&session=' +
        encodeURIComponent(
          sessionToken || ''
        );

    }


    


    document.addEventListener(
      'DOMContentLoaded',
      function() {

        const navButton =
          document.getElementById(
            'navCandidateDatabase'
          );

        if (navButton) {

          navButton.classList.add(
            'active'
          );

        }

      }
    );

  </script>
