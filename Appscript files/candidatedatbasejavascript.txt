  function loadCandidateDatabase() {
 
    const sessionToken =
  localStorage.getItem(
    'OSYNIX_ATS_SESSION'
  );
 
    if (!sessionToken) {
 
      console.error(
        'Candidate Database: No ATS session token found.'
      );
 
      return;
    }
 
 
    google.script.run
 
      .withSuccessHandler(
        function(data) {
 
          candidateDatabaseData =
            data;
 
          console.log(
            'Candidate Database loaded:',
            data
          );
 
 
          /*
           * For this first test we are only
           * populating the KPI cards.
           *
           * Table, filters and insights come
           * after the backend is verified.
           */
 
          renderCandidateDatabaseSummary(
            data
          );
          renderCandidateDatabaseInsights(
  data
);
 
      })
 
      .withFailureHandler(
  function(error) {
 
    console.error(
      'Candidate Database load failed:',
      error
    );

      'Candidate Database error message:',
      error && error.message
        ? error.message

    );

      'Candidate Database Error:\n\n' +

          : String(error)

                'class="candidate-db-loading-cell">' +
                  'Unable to load candidate database.' +
                '</td>' +
              '</tr>';

      })

        sessionToken
      );

function renderCandidateStatusInsight_(
  candidates
) {
 
  const container =
    document.getElementById(
      'candidateStatusInsight'
    );
 
  if (!container) {
    return;
  }
 
 
  const counts = {};
 
  candidates.forEach(
    function(candidate) {
 
      const status =
        String(
          candidate.talentPoolStatus ||
          'Unspecified'
        ).trim();
 
      counts[status] =
        (counts[status] || 0) + 1;
 
    }
  );
 
 
  const entries =
    Object.keys(counts)
      .map(
        function(status) {
          return {
            label: status,
            count: counts[status]
          };
        }
      )
      .sort(
        function(a, b) {
          return b.count - a.count;
        }
      );
 
 
  if (!entries.length) {
 
    container.innerHTML =
      '<div class="candidate-db-placeholder">' +
        'No talent pool status data available.' +
      '</div>';
 
    return;
 
  }
 
 
  const total =
    candidates.length || 0;
 
 
  /*
   * Build conic-gradient segments.
   * Uses the Osynix palette while keeping
   * the dashboard visually restrained.
   */
 
  const colors = [
    '#175AFF',
    '#1C1E66',
    '#31B67A',
    '#8B6CE8',
    '#E9A23B',
    '#E45D5D'
  ];
 
  let start = 0;
 
  const segments =
    entries.map(
      function(item, index) {
 
        const size =
          total
            ? (item.count / total) * 100
            : 0;
 
        const end =
          start + size;
 
        const segment =
          colors[
            index % colors.length
          ] +
          ' ' +
          start +
          '% ' +
          end +
          '%';
 
        start = end;
 
        return segment;
 
      }
    );
 
 
  const legend =
    entries.map(
      function(item, index) {
 
        const percentage =
          total
            ? Math.round(
                (item.count / total) * 100
              )
            : 0;
 
        return (
          '<div class="candidate-status-legend-row">' +
 
            '<div class="candidate-status-legend-name">' +
 
              '<span ' +
                'class="candidate-status-dot" ' +
                'style="background:' +
                  colors[
                    index % colors.length
                  ] +
                '">' +
              '</span>' +
 
              '<span>' +
                escapeCandidateDatabaseHtml_(
                  item.label
                ) +
              '</span>' +
 
            '</div>' +
 
            '<div class="candidate-status-legend-value">' +
 
              '<strong>' +
                item.count +
              '</strong>' +
 
              '<small>' +
                percentage +
                '%' +
              '</small>' +
 
            '</div>' +
 
          '</div>'
        );
 
      }
    )
    .join('');
 
 
  container.innerHTML =
 
    '<div class="candidate-status-layout">' +
 
      '<div class="candidate-status-donut-wrap">' +
 
        '<div ' +
          'class="candidate-status-donut" ' +
          'style="background:conic-gradient(' +
            segments.join(',') +
          ')">' +
 
          '<div class="candidate-status-donut-center">' +
 
            '<strong>' +
              total +
            '</strong>' +
 
            '<span>Total Talent</span>' +
 
          '</div>' +
 
        '</div>' +
 
      '</div>' +
 
      '<div class="candidate-status-legend">' +
        legend +
      '</div>' +
 
    '</div>';
 
}
function normalizeCandidateSeniority_(
  rawValue
) {
 
  const value =
    String(rawValue || '')
      .trim()
      .toLowerCase();
 
 
  if (!value) {
    return 'Others';
  }
 
 
  /*
   * Most specific / senior categories first.
   * This prevents terms such as
   * "Senior Manager" being caught as "Manager".
   */
 
  if (
    /\b(founder|co-founder|cofounder|owner|proprietor|entrepreneur)\b/i
      .test(value)
  ) {
    return 'Founder / Owner';
  }
 
 
  if (
    /\b(chief|ceo|coo|cfo|cto|chro|cmo|general manager|gm|managing director|managing partner|executive director)\b/i
      .test(value)
  ) {
    return 'Executive / GM';
  }
 
 
  if (
    /\b(head|director|vice president|vp|country manager|country head|department head|functional head)\b/i
      .test(value)
  ) {
    return 'Head / Director';
  }
 
 
  if (
    /\b(senior manager|sr\.?\s*manager|regional manager|area manager|national manager)\b/i
      .test(value)
  ) {
    return 'Senior Manager';
  }
 
 
  if (
    /\b(manager|assistant manager|deputy manager|branch manager|operations manager|project manager|program manager)\b/i
      .test(value)
  ) {
    return 'Manager';
  }
 
 
  if (
    /\b(team lead|team leader|supervisor|shift lead|shift supervisor|floor supervisor|foreman)\b/i
      .test(value)
  ) {
    return 'Team Lead / Supervisor';
  }
 
 
  if (
    /\b(senior|sr\.|specialist|consultant|advisor|analyst|engineer|accountant|executive|officer)\b/i
      .test(value)
  ) {
    return 'Senior Professional';
  }
 
 
  if (
    /\b(associate|coordinator|assistant|representative|agent|developer|designer|writer|recruiter|administrator|admin|technician)\b/i
      .test(value)
  ) {
    return 'Individual Contributor';
  }
 
 
  if (
    /\b(commander|commanding officer|colonel|major|captain|lieutenant|brigadier|military officer)\b/i
      .test(value)
  ) {
    return 'Officer / Command';
  }
 
 
  return 'Others';
 
}
 
 
function normalizeCandidateCity_(
  rawValue
) {
 
  const value =
    String(rawValue || '')
      .trim()
      .toLowerCase();
 
 
  if (!value) {
    return 'Others';
  }
 
 
  /*
   * City matching intentionally searches
   * the complete location string.
   *
   * Examples:
   * Bahria Town, Lahore -> Lahore
   * DHA Lahore          -> Lahore
   * Gulberg, Lahore     -> Lahore
   */
 
 
  const cityRules = [
 
    {
      city: 'Lahore',
      pattern:
        /\b(lahore|dha lahore|gulberg|johar town|model town|bahria town lahore|bahria orchard)\b/i
    },
 
    {
      city: 'Karachi',
      pattern:
        /\b(karachi|dha karachi|clifton|korangi|malir|gulshan-e-iqbal|gulistan-e-johar|shahrah-e-faisal)\b/i
    },
 
    {
      city: 'Islamabad',
      pattern:
        /\b(islamabad|blue area|f-\d+|g-\d+|i-\d+|e-\d+)\b/i
    },
 
    {
      city: 'Rawalpindi',
      pattern:
        /\b(rawalpindi|saddar rawalpindi|chaklala)\b/i
    },
 
    {
      city: 'Faisalabad',
      pattern:
        /\b(faisalabad)\b/i
    },
 
    {
      city: 'Multan',
      pattern:
        /\b(multan)\b/i
    },
 
    {
      city: 'Gujranwala',
      pattern:
        /\b(gujranwala)\b/i
    },
 
    {
      city: 'Sialkot',
      pattern:
        /\b(sialkot)\b/i
    },
 
    {
      city: 'Bahawalpur',
      pattern:
        /\b(bahawalpur)\b/i
    },
 
    {
      city: 'Bahawalnagar',
      pattern:
        /\b(bahawalnagar)\b/i
    },
 
    {
      city: 'Peshawar',
      pattern:
        /\b(peshawar)\b/i
    },
 
    {
      city: 'Quetta',
      pattern:
        /\b(quetta)\b/i
    },
 
    {
      city: 'Hyderabad',
      pattern:
        /\b(hyderabad)\b/i
    }
 
  ];
 
 
  for (
    let i = 0;
    i < cityRules.length;
    i++
  ) {
 
    if (
      cityRules[i].pattern.test(
        value
      )
    ) {
      return cityRules[i].city;
    }
 
  }
 
 
  return 'Others';
 
}
 
function renderCandidateSkillsInsight_(
  candidates
) {
 
  const container =
    document.getElementById(
      'candidateSkillsInsight'
    );
 
  if (!container) {
    return;
  }
 
 
  /*
   * Keep the ATS seniority taxonomy in its
   * intended progression rather than sorting
   * categories alphabetically.
   */
 
  const seniorityOrder = [
    'Individual Contributor',
    'Senior Professional',
    'Team Lead / Supervisor',
    'Manager',
    'Senior Manager',
    'Head / Director',
    'Executive / GM',
    'Founder / Owner',
    'Officer / Command'
  ];
 
 
  const counts = {};
 
  seniorityOrder.forEach(
    function(level) {
      counts[level] = 0;
    }
  );
 
 
  let unspecified = 0;
 
 
  candidates.forEach(
    function(candidate) {
 
      const seniority =
  normalizeCandidateSeniority_(
    candidate.seniorityLevel
  );
 
 
      const matchedLevel =
        seniorityOrder.find(
          function(level) {
 
            return (
              level.toLowerCase() ===
              seniority.toLowerCase()
            );
 
          }
        );
 
 
      if (matchedLevel) {
 
        counts[matchedLevel]++;
 
      } else {
 
        /*
         * Preserve unexpected historical values
         * instead of silently discarding them.
         */
 
        if (
          counts[seniority] === undefined
        ) {
          counts[seniority] = 0;
        }
 
        counts[seniority]++;
 
      }
 
    }
  );
 
 
  const entries = [];
 
 
  seniorityOrder.forEach(
    function(level) {
 
      if (counts[level] > 0) {
 
        entries.push({
          label: level,
          count: counts[level]
        });
 
      }
 
    }
  );
 
 
  /*
   * Include any historical/non-standard values
   * after the controlled taxonomy.
   */
 
  Object.keys(counts)
    .filter(
      function(level) {
 
        return (
          seniorityOrder.indexOf(level) === -1 &&
          counts[level] > 0
        );
 
      }
    )
    .forEach(
      function(level) {
 
        entries.push({
          label: level,
          count: counts[level]
        });
 
      }
    );
 
 
 
  if (!entries.length) {
 
    container.innerHTML =
      '<div class="candidate-db-placeholder">' +
        'No seniority data available.' +
      '</div>';
 
    return;
 
  }
 
 
  const total =
    candidates.length || 1;
 
 
  const maxCount =
  Math.max.apply(
    null,
    entries.map(
      function(item) {
        return item.count;
      }
    )
  ) || 1;
 
 
container.innerHTML =
  '<div class="candidate-vertical-chart">' +
 
    entries.map(
      function(item) {
 
        const height =
          Math.max(
            8,
            Math.round(
              (
                item.count /
                maxCount
              ) * 100
            )
          );
 
 
        return (
          '<div class="candidate-vertical-item">' +
 
            '<div class="candidate-vertical-value">' +
              item.count +
            '</div>' +
 
            '<div class="candidate-vertical-track">' +
 
              '<div ' +
                'class="candidate-vertical-bar" ' +
'style="height:' +
  height +
'%;background:' +
  getCandidateSeniorityColor_(
    item.label
  ) +
'">' +
              '</div>' +
 
            '</div>' +
 
            '<div ' +
              'class="candidate-vertical-label" ' +
              'title="' +
                escapeCandidateDatabaseHtml_(
                  item.label
                ) +
              '">' +
 
              escapeCandidateDatabaseHtml_(
                item.label
              ) +
 
            '</div>' +
 
          '</div>'
        );
 
      }
    ).join('') +
 
  '</div>';
 
}

function renderCandidateLocationsInsight_(
  candidates
) {
 
  const container =
    document.getElementById(
      'candidateLocationsInsight'
    );
 
  if (!container) {
    return;
  }
 
 
  const counts = {};
 
 
  candidates.forEach(
    function(candidate) {
 
      const location =
  normalizeCandidateCity_(
    candidate.location
  );
 
 
      const key =
        location.toLowerCase();
 
 
      if (!counts[key]) {
 
        counts[key] = {
          label: location,
          count: 0
        };
 
      }
 
 
      counts[key].count++;
 
    }
  );
 
 
  const topLocations =
    Object.keys(counts)
      .map(
        function(key) {
          return counts[key];
        }
      )
      .sort(
        function(a, b) {
 
          if (b.count !== a.count) {
            return b.count - a.count;
          }
 
          return a.label.localeCompare(
            b.label
          );
 
        }
      )
      .slice(0, 6);
 
 
  if (!topLocations.length) {
 
    container.innerHTML =
      '<div class="candidate-db-placeholder">' +
        'No location data available.' +
      '</div>';
 
    return;
 
  }
 
 
  const highestCount =
    topLocations[0].count || 1;
 
 
  const maxCount =
  Math.max.apply(
    null,
    topLocations.map(
      function(item) {
        return item.count;
      }
    )
  ) || 1;
 
 
container.innerHTML =
  '<div class="candidate-vertical-chart">' +
 
    topLocations.map(
      function(item) {
 
        const height =
          Math.max(
            8,
            Math.round(
              (
                item.count /
                maxCount
              ) * 100
            )
          );
 
 
        return (
          '<div class="candidate-vertical-item">' +
 
            '<div class="candidate-vertical-value">' +
              item.count +
            '</div>' +
 
            '<div class="candidate-vertical-track">' +
 
              '<div ' +
                'class="candidate-vertical-bar" ' +
'style="height:' +
  height +
'%;background:' +
  getCandidateLocationColor_(
    item.label
  ) +
'">' +
              '</div>' +
 
            '</div>' +
 
            '<div ' +
              'class="candidate-vertical-label" ' +
              'title="' +
                escapeCandidateDatabaseHtml_(
                  item.label
                ) +
              '">' +
 
              escapeCandidateDatabaseHtml_(
                item.label
              ) +
 
            '</div>' +
 
          '</div>'
        );
 
      }
    ).join('') +
 
  '</div>';
 
}
function getCandidateLocationColor_(
  location
) {
 
  if (
    String(location).toLowerCase() ===
    'others'
  ) {
    return '#94A3B8';
  }
 
 
  const palette = [
    '#175AFF',
    '#3B6EFF',
    '#5865E8',
    '#1C1E66',
    '#7357D9',
    '#4E83E6'
  ];
 
 
  let hash = 0;
 
  const value =
    String(location || '');
 
 
  for (
    let i = 0;
    i < value.length;
    i++
  ) {
 
    hash =
      value.charCodeAt(i) +
      (
        (hash << 5) -
        hash
      );
 
  }
 
 
  return palette[
    Math.abs(hash) %
    palette.length
  ];
 
}
 
function escapeCandidateDatabaseHtml_(
  value
) {
 
  return String(
    value === undefined ||
    value === null
      ? ''
      : value
  )
    .replace(
      /&/g,
      '&amp;'
    )
    .replace(
      /</g,
      '&lt;'
    )
    .replace(
      />/g,
      '&gt;'
    )
    .replace(
      /"/g,
      '&quot;'
    )
    .replace(
      /'/g,
      '&#039;'
    );
 
}

function getCandidateLifecycleState_(
  candidate
) {
 
  const assessment =
    candidate &&
    candidate.latestAssessment
      ? candidate.latestAssessment
      : {};
 
 
  const stage =
    String(
      assessment.recruitmentStage || ''
    )
      .trim()
      .toLowerCase();
 
 
  const interviewStatus =
    String(
      assessment.interviewStatus || ''
    )
      .trim()
      .toLowerCase();
 
 
  const clientStatus =
    String(
      assessment.clientStatus || ''
    )
      .trim()
      .toLowerCase();
 
 
  const talentPoolStatus =
    String(
      candidate &&
      candidate.talentPoolStatus
        ? candidate.talentPoolStatus
        : ''
    )
      .trim()
      .toLowerCase();
 
 
  /*
   * 1. HIRED / PLACED
   * Highest-priority successful final outcome.
   */
 
  if (
    clientStatus === 'hired' ||
    stage === 'hired' ||
    talentPoolStatus === 'placed'
  ) {
    return 'hired';
  }
 
 
  /*
   * 2. REJECTED
   * Final negative outcome overrides all earlier stages.
   */
 
  if (
    clientStatus === 'client rejected' ||
    stage === 'rejected'
  ) {
    return 'rejected';
  }
 
 
  /*
   * 3. PENDING DECISION
   *
   * Offer is intentionally treated as pending.
   * Completed interviews also move into Pending Decision.
   */
 
  if (
    clientStatus === 'offer' ||
    stage === 'offer sent' ||
    interviewStatus === 'completed' ||
    clientStatus === 'submitted' ||
    clientStatus === 'under review' ||
    clientStatus === 'client shortlisted'
  ) {
    return 'pending';
  }
 
 
  /*
   * 4. IN INTERVIEW
   *
   * Candidate is actively progressing through an
   * interview stage and no downstream outcome has
   * already overridden it.
   */
 
  if (
    stage === 'technical interview' ||
    stage === 'final interview' ||
    stage === 'client interview' ||
    clientStatus === 'client interview'
  ) {
    return 'interview';
  }
 
 
  /*
   * 5. ON HOLD
   */
 
  if (
    stage === 'hold'
  ) {
    return 'hold';
  }
 
 
  /*
   * 6. SHORTLISTED
   */
 
  if (
    stage === 'shortlisted'
  ) {
    return 'shortlisted';
  }
 
 
  /*
   * Candidate does not currently belong to one of
   * the operational Candidate Database KPI queues.
   */
 
  return '';
}

      );

  );
 
candidateReactivePositionFilter = '';

updateCandidateReactiveFilterLabels_();

updateCandidateReactiveHeading_(type);

function renderCandidateReactivePage_() {
 
  const totalRecords =
    candidateReactiveFilteredCandidates.length;
 
  const totalPages =
    Math.max(
      1,
      Math.ceil(
        totalRecords /
        candidateReactivePageSize
      )
    );
 
 
  if (
    candidateReactiveCurrentPage >
    totalPages
  ) {
    candidateReactiveCurrentPage =
      totalPages;
  }
 
 
  const startIndex =
    (
      candidateReactiveCurrentPage - 1
    ) *
    candidateReactivePageSize;
 
 
  const endIndex =
    startIndex +
    candidateReactivePageSize;
 
 
  const pageRecords =
    candidateReactiveFilteredCandidates.slice(
      startIndex,
      endIndex
    );
 
 
  renderCandidateReactiveTable_(
    pageRecords
  );
 
 
  renderCandidateReactivePagination_(
    totalRecords,
    totalPages,
    startIndex,
    pageRecords.length
  );
 
}
 
 
function renderCandidateReactivePagination_(
  totalRecords,
  totalPages,
  startIndex,
  visibleCount
) {
 
  const container =
    document.getElementById(
      'candidateReactivePagination'
    );
 
 
  const count =
    document.getElementById(
      'candidateReactiveCount'
    );
 
 
  if (count) {
 
    if (!totalRecords) {
 
      count.textContent =
        '0 candidates';
 
    } else {
 
      const firstRecord =
        startIndex + 1;
 
      const lastRecord =
        startIndex + visibleCount;
 
      count.textContent =
        'Showing ' +
        firstRecord +
        '–' +
        lastRecord +
        ' of ' +
        totalRecords +
        (
          totalRecords === 1
            ? ' candidate'
            : ' candidates'
        );
 
    }
 
  }
 
 
  if (!container) {
    return;
  }
 
 
  if (
    totalPages <= 1 ||
    totalRecords === 0
  ) {
 
    container.innerHTML = '';
    return;
 
  }
 
 
  let html = '';
 
 
  html +=
    '<button ' +
      'type="button" ' +
      'class="candidate-reactive-page-btn" ' +
      (
        candidateReactiveCurrentPage === 1
          ? 'disabled '
          : ''
      ) +
      'onclick="changeCandidateReactivePage_(' +
        (
          candidateReactiveCurrentPage - 1
        ) +
      ')">' +
      'Previous' +
    '</button>';
 
 
  for (
    let page = 1;
    page <= totalPages;
    page++
  ) {
 
    html +=
      '<button ' +
        'type="button" ' +
        'class="candidate-reactive-page-btn' +
        (
          page ===
          candidateReactiveCurrentPage
            ? ' active'
            : ''
        ) +
        '" ' +
        'onclick="changeCandidateReactivePage_(' +
          page +
        ')">' +
        page +
      '</button>';
 
  }
 
 
  html +=
    '<button ' +
      'type="button" ' +
      'class="candidate-reactive-page-btn" ' +
      (
        candidateReactiveCurrentPage ===
        totalPages
          ? 'disabled '
          : ''
      ) +
      'onclick="changeCandidateReactivePage_(' +
        (
          candidateReactiveCurrentPage + 1
        ) +
      ')">' +
      'Next' +
    '</button>';
 
 
  container.innerHTML = html;
 
}

function filterCandidateReactiveView_() {
 
  const search =
    document.getElementById(
      'candidateReactiveSearch'
    );
 
 
  const query =
    search
      ? String(search.value || '')
          .trim()
          .toLowerCase()
      : '';
 
 
  const filtered =
    candidateReactiveCandidates.filter(
      function(candidate) {
 
        const assessment =
          candidate.latestAssessment || {};
 
 
        const position =
          String(
            assessment.position || ''
          )
            .trim()
            .toLowerCase();
 
 
        const decision =
          String(
            assessment.decision || ''
          )
            .trim()
            .toLowerCase();
 
 
        const stage =
          String(
            assessment.recruitmentStage || ''
          )
            .trim()
            .toLowerCase();
 
 
        /*
         * POSITION FILTER
         */
 
        if (
          candidateReactivePositionFilter &&
          position !==
            candidateReactivePositionFilter
              .toLowerCase()
        ) {
          return false;
        }
 
 
        /*
         * AI DECISION FILTER
         */
 
        if (
          candidateReactiveDecisionFilter &&
          decision !==
            candidateReactiveDecisionFilter
              .toLowerCase()
        ) {
          return false;
        }
 
 
        /*
         * RECRUITMENT STAGE FILTER
         */
 
        if (
          candidateReactiveStageFilter &&
          stage !==
            candidateReactiveStageFilter
              .toLowerCase()
        ) {
          return false;
        }
 
 
        /*
         * FREE-TEXT SEARCH
         */
 
        if (query) {
 
          const searchable =
            [
              candidate.candidateName,
              candidate.currentDesignation,
              candidate.currentCompany,
              candidate.location,
              candidate.coreSkills,
              assessment.refNo,
              assessment.refNumber,
              assessment.ref,
              assessment.position,
              assessment.client,
              assessment.decision,
              assessment.recruitmentStage
            ]
              .join(' ')
              .toLowerCase();
 
 
          if (
            searchable.indexOf(query) === -1
          ) {
            return false;
          }
 
        }
 
 
        return true;
 
      }
    );
 
 
  candidateReactiveFilteredCandidates =
    filtered;
 
 
  candidateReactiveCurrentPage = 1;
 
 
  renderCandidateReactivePage_();
 
}

function getCandidateReactiveFilterValues_(
  type
) {
 
  const values =
    new Set();
 
 
  candidateReactiveCandidates.forEach(
    function(candidate) {
 
      const assessment =
        candidate.latestAssessment || {};
 
 
      let value = '';
 
 
      if (type === 'position') {
 
        value =
          String(
            assessment.position || ''
          ).trim();
 
      }
 
 
      if (type === 'decision') {
 
        value =
          String(
            assessment.decision || ''
          ).trim();
 
      }
 
 
      if (type === 'stage') {
 
        value =
          String(
            assessment.recruitmentStage || ''
          ).trim();
 
      }
 
 
      if (value) {
        values.add(value);
      }
 
    }
  );
 
 
  return Array.from(values)
    .sort(
      function(a, b) {
        return a.localeCompare(b);
      }
    );
 
}

  if (type === 'stage') {

  }
 
 
  return '';
 
}

function loadCandidateProfile_(candidateId, refNo) {
  const sessionToken = localStorage.getItem('OSYNIX_ATS_SESSION');
 
  if (!sessionToken) {
    showCandidateProfileError_('ATS session token was not found.');
    return;
  }
 
  google.script.run
    .withSuccessHandler(function(data) {
      currentCandidateProfileData_ = data || null;
      currentCandidateProfileId_ = candidateId || '';
      currentCandidateProfileRef_ =
        data && data.selectedAssessment && data.selectedAssessment.refNo
          ? String(data.selectedAssessment.refNo)
          : String(refNo || '');
      renderCandidateProfile_(data);
    })
    .withFailureHandler(function(error) {
      console.error('Candidate Profile load failed:', error);
      showCandidateProfileError_(
        error && error.message ? error.message : String(error)
      );
    })
    .getCandidateProfile(sessionToken, candidateId, refNo || '');
}

function candidateProfileDurationFromRange_(rangeText) {
  const text = String(rangeText || '').trim();
  if (!text) return '';
 
  const months = {
    jan: 0, january: 0, feb: 1, february: 1, mar: 2, march: 2,
    apr: 3, april: 3, may: 4, jun: 5, june: 5, jul: 6, july: 6,
    aug: 7, august: 7, sep: 8, sept: 8, september: 8, oct: 9, october: 9,
    nov: 10, november: 10, dec: 11, december: 11
  };
 
  function parsePoint_(part, isEnd) {
    const value = String(part || '').trim().toLowerCase();
    if (/^(present|current|now)$/.test(value)) return new Date();
 
    const m = value.match(/^([a-z]+)\s+((?:19|20)\d{2})$/i);
    if (m && months[m[1].toLowerCase()] !== undefined) {
      return new Date(Number(m[2]), months[m[1].toLowerCase()], 1);
    }
 
    const y = value.match(/^((?:19|20)\d{2})$/);
    if (y) return new Date(Number(y[1]), isEnd ? 11 : 0, 1);
 
    return null;
  }
 
  const parts = text.split(/\s*[–—-]\s*/);
  if (parts.length !== 2) return '';
 
  const from = parsePoint_(parts[0], false);
  const to = parsePoint_(parts[1], true);
  if (!from || !to || to < from) return '';
 
  let totalMonths = (to.getFullYear() - from.getFullYear()) * 12 + (to.getMonth() - from.getMonth());
  if (/present|current|now/i.test(parts[1])) totalMonths += 1;
  if (totalMonths < 1) return '';
 
  const years = Math.floor(totalMonths / 12);
  const remainingMonths = totalMonths % 12;
  const bits = [];
  if (years) bits.push(years + ' yr' + (years === 1 ? '' : 's'));
  if (remainingMonths) bits.push(remainingMonths + ' mo' + (remainingMonths === 1 ? '' : 's'));
  return bits.join(' ');
}
 
function renderCandidateProfileEmployment_(value) {
  const container = document.getElementById('candidateProfileEmploymentHistory');
  if (!container) return;
 
  if (Array.isArray(value)) {
    const items = value.filter(function(item) {
      return item && typeof item === 'object' &&
        (item.company_name || item.designation || item.start_date || item.end_date || item.location);
    });
 
    if (!items.length) {
      container.textContent = '—';
      return;
    }
 
    container.innerHTML = '<div class="candidate-profile-timeline">' + items.map(function(item) {
      const company = String(item.company_name || '').trim();
      const designation = String(item.designation || '').trim();
      const startDate = String(item.start_date || '').trim();
      const endDate = String(item.end_date || '').trim();
      const location = String(item.location || '').trim();
      const timeline = startDate && endDate ? startDate + ' – ' + endDate : (startDate || endDate);
      const meta = [designation, location].filter(Boolean).join(' • ');
 
      return '<div class="candidate-profile-timeline-item">' +
        '<div class="candidate-profile-timeline-heading">' +
          '<div class="candidate-profile-timeline-title">' + escapeCandidateDatabaseHtml_(company || designation || 'Employment') + '</div>' +
          (timeline ? '<div class="candidate-profile-timeline-date">' + escapeCandidateDatabaseHtml_(timeline) + '</div>' : '') +
        '</div>' +
        (company && designation ? '<div class="candidate-profile-timeline-meta">' + escapeCandidateDatabaseHtml_(designation) + '</div>' : '') +
        (location ? '<div class="candidate-profile-timeline-copy">' + escapeCandidateDatabaseHtml_(location) + '</div>' : '') +
      '</div>';
    }).join('') + '</div>';
    return;
  }
 
  const sentences = splitCandidateProfileSentences_(value);
  if (!sentences.length) {
    container.textContent = '—';
    return;
  }
 
  const entries = [];
  sentences.forEach(function(sentence) {
    const isNewRole = /\s[—–-]\s/.test(sentence) ||
      (/\b(present|current)\b/i.test(sentence) && /\b(19|20)\d{2}\b/.test(sentence));
 
    if (isNewRole || !entries.length) {
      entries.push({ heading: sentence, details: [] });
    } else {
      entries[entries.length - 1].details.push(sentence);
    }
  });
 
  container.innerHTML = '<div class="candidate-profile-timeline">' + entries.map(function(entry) {
    const parts = entry.heading.split(/\s[—–-]\s/, 2);
    const employer = parts.length > 1 ? parts[0].trim() : entry.heading.trim();
    let roleMeta = parts.length > 1 ? parts[1].trim() : '';
 
    let timeline = '';
    let duration = '';
 
    const dateRangeMatch = roleMeta.match(/(?:,\s*)?((?:(?:Jan(?:uary)?|Feb(?:ruary)?|Mar(?:ch)?|Apr(?:il)?|May|Jun(?:e)?|Jul(?:y)?|Aug(?:ust)?|Sep(?:t(?:ember)?)?|Oct(?:ober)?|Nov(?:ember)?|Dec(?:ember)?)\s+)?(?:19|20)\d{2}\s*[–—-]\s*(?:(?:(?:Jan(?:uary)?|Feb(?:ruary)?|Mar(?:ch)?|Apr(?:il)?|May|Jun(?:e)?|Jul(?:y)?|Aug(?:ust)?|Sep(?:t(?:ember)?)?|Oct(?:ober)?|Nov(?:ember)?|Dec(?:ember)?)\s+)?(?:19|20)\d{2}|present|current))\.?$/i);
    const durationMatch = !dateRangeMatch
      ? roleMeta.match(/(?:,\s*)?(\d+(?:\.\d+)?\s+(?:years?|yrs?|months?|mos?))(?=;|\.|$)/i)
      : null;
 
    if (dateRangeMatch) {
      timeline = dateRangeMatch[1].trim().replace(/\.$/, '');
      duration = candidateProfileDurationFromRange_(timeline);
      roleMeta = roleMeta.replace(dateRangeMatch[0], '').replace(/[,\s]+$/, '').trim();
    } else if (durationMatch) {
      duration = durationMatch[1].trim();
      roleMeta = roleMeta.replace(durationMatch[0], '').replace(/[,\s;]+$/, '').trim();
    }
 
    const semicolonParts = roleMeta.split(/\s*;\s*/);
    const designation = (semicolonParts.shift() || '').trim().replace(/\.$/, '');
    const inlineDetail = semicolonParts.join('; ').trim();
    const detailText = [inlineDetail].concat(entry.details).filter(Boolean).join(' ');
    const timelineLabel = [timeline, duration].filter(Boolean).join(' • ');
 
    return '<div class="candidate-profile-timeline-item">' +
      '<div class="candidate-profile-timeline-heading">' +
        '<div class="candidate-profile-timeline-title">' + escapeCandidateDatabaseHtml_(employer) + '</div>' +
        (timelineLabel ? '<div class="candidate-profile-timeline-date">' + escapeCandidateDatabaseHtml_(timelineLabel) + '</div>' : '') +
      '</div>' +
      (designation ? '<div class="candidate-profile-timeline-meta">' + escapeCandidateDatabaseHtml_(designation) + '</div>' : '') +
      (detailText ? '<div class="candidate-profile-timeline-copy">' + escapeCandidateDatabaseHtml_(detailText) + '</div>' : '') +
      '</div>';
  }).join('') + '</div>';
}

function candidateAssessmentNumber_(value, scale) {
  const raw = String(value === null || value === undefined ? '' : value).replace('%', '').trim();
 
  // Blank historical criterion values must remain blank.
  // Number('') returns 0 in JavaScript, which previously caused
  // unstored historical scores/confidence to display as 0/10.
  if (raw === '') return null;
 
  const number = Number(raw);
  if (!isFinite(number)) return null;
  return Math.max(0, Math.min(scale, number));
}
 
function setCandidateAssessmentDonut_(id, value, scale, labelId, suffix) {
  const donut = document.getElementById(id);
  const label = document.getElementById(labelId);
  const numeric = candidateAssessmentNumber_(value, scale);
  if (!donut || !label) return;
 
  if (numeric === null) {
    donut.style.setProperty('--score-angle', '0deg');
    donut.classList.add('is-empty');
    label.textContent = '—';
    return;
  }
 
  const percent = Math.round((numeric / scale) * 100);
  donut.style.setProperty('--score-angle', (percent * 3.6) + 'deg');
  donut.classList.remove('is-empty');
  donut.classList.toggle('is-strong', percent >= 70);
  donut.classList.toggle('is-medium', percent >= 45 && percent < 70);
  donut.classList.toggle('is-low', percent < 45);
  label.textContent = numeric + suffix;
}

function renderCandidateAssessmentCriteria_(criteria) {
  const container = document.getElementById('candidateAssessmentCriteria');
  if (!container) return;
 
  if (!criteria.length) {
    container.innerHTML =
      '<div class="candidate-assessment-empty">' +
        '<strong>Detailed criterion results are not stored for this earlier assessment.</strong>' +
        '<span>Future assessments saved after this update will retain the full weighted criterion review here.</span>' +
      '</div>';
    return;
  }
 
  container.innerHTML = criteria.map(function(item, index) {
    const score = candidateAssessmentNumber_(item.score10 !== undefined ? item.score10 : item.score_10, 10);
    const confidence = candidateAssessmentNumber_(item.evidenceConfidence10 !== undefined ? item.evidenceConfidence10 : item.evidence_confidence_10, 10);
    const weight = Number(item.weight || 0);
    const historicalFrameworkOnly = Boolean(item.historicalFrameworkOnly);
    const status = historicalFrameworkOnly
      ? 'Framework'
      : (String(item.evidenceStatus || item.status || '').trim() || (score >= 6 ? 'Pass' : score >= 4 ? 'Partial' : 'Fail'));
    const statusClass = historicalFrameworkOnly
      ? 'is-framework'
      : (status.toLowerCase().indexOf('pass') >= 0 ? 'is-pass' : status.toLowerCase().indexOf('partial') >= 0 ? 'is-partial' : 'is-fail');
    const knockout = String(item.knockout || '').toLowerCase() === 'yes';
    const scorePercent = score === null ? 0 : score * 10;
 
    return '<article class="candidate-assessment-criterion">' +
      '<div class="candidate-assessment-criterion-top">' +
        '<div><span class="candidate-assessment-criterion-index">' + String(index + 1).padStart(2, '0') + '</span>' +
        '<strong>' + escapeCandidateDatabaseHtml_(item.criterion || 'Criterion') + '</strong></div>' +
        '<div class="candidate-assessment-criterion-tags">' +
          (knockout ? '<span class="is-knockout">Must-Have</span>' : '') +
          '<span>' + escapeCandidateDatabaseHtml_(String(weight)) + '% Weight</span>' +
          '<span class="' + statusClass + '">' + escapeCandidateDatabaseHtml_(status) + '</span>' +
        '</div>' +
      '</div>' +
      '<div class="candidate-assessment-criterion-meter"><i style="width:' + scorePercent + '%"></i></div>' +
      '<div class="candidate-assessment-criterion-metrics">' +
        '<span>Score <b>' + (score === null ? '—' : score + '/10') + '</b></span>' +
        '<span>Evidence Confidence <b>' + (confidence === null ? '—' : confidence + '/10') + '</b></span>' +
      '</div>' +
      (item.evidenceFound || item.evidence_found ? '<div class="candidate-assessment-criterion-copy"><span>Evidence Found</span><p>' + escapeCandidateDatabaseHtml_(item.evidenceFound || item.evidence_found) + '</p></div>' : '') +

function saveCandidateProfile_() {
  const data = currentCandidateProfileData_ || {};
  const assessment = data.selectedAssessment || {};
  const profile = data.profile || {};
  const sessionToken = localStorage.getItem('OSYNIX_ATS_SESSION');
 
  if (!sessionToken || !profile.candidateId || !assessment.refNo) {
    setCandidateProfileText_('candidateProfileSaveStatus', 'Select an assessed application before saving.');
    return;
  }
 
  const button = document.getElementById('candidateProfileSaveBtn');
  if (button) button.disabled = true;
  setCandidateProfileText_('candidateProfileSaveStatus', 'Saving recruitment update...');
 
  const payload = {
    candidateId: profile.candidateId,
    refNo: assessment.refNo,
    recruitmentStage: getCandidateProfileInputValue_('candidateProfileRecruitmentStage'),
    interviewStatus: getCandidateProfileInputValue_('candidateProfileInterviewStatus'),
    clientStatus: getCandidateProfileInputValue_('candidateProfileClientStatus'),
    expectedSalary: getCandidateProfileInputValue_('candidateProfileExpectedSalary'),
    availability: getCandidateProfileInputValue_('candidateProfileAvailability'),
    recruiterNotes: getCandidateProfileInputValue_('candidateProfileRecruiterNotes')
  };
 
  google.script.run
    .withSuccessHandler(function(updatedData) {
      currentCandidateProfileData_ = updatedData || null;
      renderCandidateProfile_(updatedData);
      setCandidateProfileText_('candidateProfileSaveStatus', 'Saved successfully.');
      if (button) button.disabled = false;
      loadCandidateDatabase();
    })
    .withFailureHandler(function(error) {
      console.error('Candidate Profile save failed:', error);
      setCandidateProfileText_(
        'candidateProfileSaveStatus',
        'Unable to save: ' + (error && error.message ? error.message : String(error))
      );
      if (button) button.disabled = false;
    })
    .saveCandidateProfileOperational(sessionToken, payload);
}

function enhanceCandidateProfileOverviewApplications_(history, selectedRefNo) {
  const rightPanel = document.querySelector('.candidate-profile-overview-right');
  if (!rightPanel) return;
 
  const oldStack = rightPanel.querySelector('.candidate-profile-application-stack');
  if (oldStack) oldStack.remove();
 
  const visibleCards = Array.from(rightPanel.querySelectorAll(':scope > .candidate-profile-context-card'))
    .filter(function(card) { return card.style.display !== 'none'; });
 
  visibleCards.forEach(function(card) {
    if (card.classList.contains('candidate-profile-collapsible-card')) return;
 
    const body = document.createElement('div');
    body.className = 'candidate-profile-collapsible-body';
 
    while (card.firstChild) body.appendChild(card.firstChild);
 
    
    const duplicateHeading = body.querySelector(
      ':scope > h3, :scope > .candidate-profile-card-heading'
    );
    if (
      duplicateHeading &&
      /latest\s*\/\s*selected\s+position/i.test(
        String(duplicateHeading.textContent || '')
      )
    ) {
      duplicateHeading.remove();
    }
 
    const header = document.createElement('button');
    header.type = 'button';
    header.className = 'candidate-profile-collapsible-header';
    header.setAttribute('aria-expanded', 'true');
    header.innerHTML =
      '<span>Latest / Selected Position</span>' +
      '<span class="candidate-profile-collapse-icon">−</span>';
    header.onclick = function() {
      const expanded = card.classList.toggle('is-collapsed') === false;
      header.setAttribute('aria-expanded', expanded ? 'true' : 'false');
      const icon = header.querySelector('.candidate-profile-collapse-icon');
      if (icon) icon.textContent = expanded ? '−' : '+';
    };
 
    card.classList.add('candidate-profile-collapsible-card');
    card.appendChild(header);
    card.appendChild(body);
  });
 
  const items = Array.isArray(history) ? history : [];
  const previous = items.filter(function(item) {
    return String(item.refNo || '') !== String(selectedRefNo || '');
  });
  if (!previous.length) return;
 
  const stack = document.createElement('div');
  stack.className = 'candidate-profile-application-stack';
 
  const heading = document.createElement('div');
  heading.className = 'candidate-profile-application-stack-title';
  heading.textContent = 'Previous Applications';
  stack.appendChild(heading);
 
  const scroll = document.createElement('div');
  scroll.className = 'candidate-profile-application-scroll';
 
  previous.forEach(function(item) {
    const card = document.createElement('div');
    card.className = 'candidate-profile-application-card is-collapsed';
 
    const button = document.createElement('button');
    button.type = 'button';
    button.className = 'candidate-profile-application-card-header';
    button.setAttribute('aria-expanded', 'false');
    button.innerHTML =
      '<span class="candidate-profile-application-card-main">' +
        '<strong>' + escapeCandidateDatabaseHtml_(item.position || 'Position') + '</strong>' +
        '<small>' +
          escapeCandidateDatabaseHtml_(
            ['Ref # ' + (item.refNo || '—'), item.client || '', item.recruitmentStage || '—']
              .filter(Boolean).join(' • ')
          ) +
        '</small>' +
      '</span>' +
      '<span class="candidate-profile-collapse-icon">+</span>';
 
    const body = document.createElement('div');
    body.className = 'candidate-profile-application-card-body';
    body.innerHTML =
      '<div><span>ATS Match</span><strong>' + escapeCandidateDatabaseHtml_(formatCandidateProfilePercent_(item.atsMatch)) + '</strong></div>' +
      '<div><span>AI Decision</span><strong>' + escapeCandidateDatabaseHtml_(item.decision || '—') + '</strong></div>' +
      '<div><span>Recruitment Stage</span><strong>' + escapeCandidateDatabaseHtml_(item.recruitmentStage || '—') + '</strong></div>' +
      '<button type="button" class="candidate-profile-application-open-btn">Open this application</button>';
 
    button.onclick = function() {
      const collapsed = card.classList.toggle('is-collapsed');
      button.setAttribute('aria-expanded', collapsed ? 'false' : 'true');
      const icon = button.querySelector('.candidate-profile-collapse-icon');
      if (icon) icon.textContent = collapsed ? '+' : '−';
    };
 
    const openButton = body.querySelector('.candidate-profile-application-open-btn');
    if (openButton) {
      openButton.onclick = function() {
        selectCandidateProfileAssessment_(item.refNo || '');
      };
    }
 
    card.appendChild(button);
    card.appendChild(body);
    scroll.appendChild(card);
  });
 
  stack.appendChild(scroll);
  rightPanel.appendChild(stack);
}

function updateCandidateReactiveStickyHeader_() {
 
  const stickyHeader =
    document.getElementById(
      'candidateReactiveStickyHeader'
    );
 
  const table =
    document.querySelector(
      '#candidateReactiveSection .candidate-db-table'
    );
 
  const normalHeader =
    document.querySelector(
      '#candidateReactiveSection .candidate-db-table thead'
    );
 
 
  if (
    !stickyHeader ||
    !table ||
    !normalHeader
  ) {
    return;
  }
 
 
  const section =
    document.getElementById(
      'candidateReactiveSection'
    );
 
 
  if (
    !section ||
    section.style.display === 'none'
  ) {
 
    stickyHeader.classList.remove(
      'is-visible'
    );
 
    return;
 
  }
 
 
  const tableRect =
    table.getBoundingClientRect();
 
  const headerRect =
    normalHeader.getBoundingClientRect();
 
 
  /*
   * Main ATS navigation occupies approximately
   * the first 66px of the viewport.
   *
   * Show the duplicate column bar after the normal
   * table heading scrolls underneath navigation,
   * and hide it again after the table has ended.
   */
 
  const navigationBottom = 66;
 
  const shouldShow =
    headerRect.bottom <= navigationBottom &&
    tableRect.bottom > navigationBottom + 46;
 
 
  stickyHeader.classList.toggle(
    'is-visible',
    shouldShow
  );
 
}
